-- ===================================================================== -- AUDIT BKCO — contexte d'un écart, étape 2 : les opérations du créneau -- -- Rassemble en une seule liste tout ce qui s'est passé sur une caisse -- pendant une session : corrections, annulations, remises, commandes -- ouvertes, repas employés. Chaque opération est rattachée par la CAISSE -- et la FENÊTRE HORAIRE de la session, jamais par la journée. -- -- Cette vue n'interprète rien. Elle répond à une seule question : que -- s'est-il passé sur cette caisse au moment où l'argent a manqué ? Le -- fait qu'il ne se soit rien passé est une réponse, souvent la plus -- parlante. -- ===================================================================== -- Index de rattachement. Sans eux, chaque ouverture de fiche balaie les -- tables entières — sur une année et sept restaurants, c'est plusieurs -- secondes par clic. create index if not exists idx_corr_caisse on corrections_lignes (restaurant_id, caisse, date_reelle); create index if not exists idx_annul_caisse on annulations_lignes (restaurant_id, caisse, date_neg); create index if not exists idx_remise_caisse on remises_lignes (restaurant_id, caisse, date_reelle); create index if not exists idx_co_caisse on co_lignes (restaurant_id, caisse, date_reelle); create index if not exists idx_repas_caisse on repas_employes (restaurant_id, caisse, date_heure); drop view if exists v_operations_session; create view v_operations_session as -- Corrections : un ticket modifié après coup. Le montant avant correction -- dit ce qui a été retiré de la vente. select b.id as session_id, b.restaurant_id, 'CORRECTION'::text as type, c.date_reelle as horodate, round(c.montant::numeric, 2) as montant, normaliser_identite(c.vendeur_code) as qui, null::text as valide_par, case when c.montant_avant is not null then 'ticket ramené de ' || round(c.montant_avant::numeric, 2) || ' €' else null end as libelle, c.num_ticket as ticket from v_bornes_session b join corrections_lignes c on c.restaurant_id = b.restaurant_id and c.caisse = b.caisse and c.date_reelle > b.debut_session and c.date_reelle <= b.fin_session union all -- Annulations : le ticket est supprimé. On garde qui a vendu et qui a -- validé l'annulation, la distinction ayant du sens. select b.id, b.restaurant_id, 'ANNULATION', a.date_neg, round(a.montant::numeric, 2), normaliser_identite(a.vendeur_pos), normaliser_identite(coalesce(a.valide_par_neg, a.manager_neg)), case when a.delai_heures is not null then 'annulé ' || round(a.delai_heures::numeric, 1) || ' h après le ticket' else null end, a.num_ticket_pos from v_bornes_session b join annulations_lignes a on a.restaurant_id = b.restaurant_id and a.caisse = b.caisse and a.date_neg > b.debut_session and a.date_neg <= b.fin_session union all -- Remises : le libellé est l'information utile, c'est lui qui distingue -- une opération nationale d'un geste local. select b.id, b.restaurant_id, 'REMISE', r.date_reelle, round(r.montant_remise::numeric, 2), normaliser_identite(r.vendeur_code), normaliser_identite(r.manager_code), r.libelle_remise, r.num_ticket from v_bornes_session b join remises_lignes r on r.restaurant_id = b.restaurant_id and r.caisse = b.caisse and r.date_reelle > b.debut_session and r.date_reelle <= b.fin_session union all -- Commandes ouvertes : la date et l'heure sont séparées dans l'export, -- on les recompose pour les situer dans la session. select b.id, b.restaurant_id, 'COMMANDE_OUVERTE', ((co.date_reelle + coalesce(co.heure, time '23:59')) at time zone 'UTC'), round(co.montant_ttc::numeric, 2), normaliser_identite(co.vendeur_code), normaliser_identite(co.annule_par), case when upper(coalesce(co.annule_par, '')) = 'AUTOMATIQUE' then 'annulée par le système, sans traçabilité' else 'non finalisée' end, co.num_orb from v_bornes_session b join co_lignes co on co.restaurant_id = b.restaurant_id and co.caisse = b.caisse and ((co.date_reelle + coalesce(co.heure, time '23:59')) at time zone 'UTC') > b.debut_session and ((co.date_reelle + coalesce(co.heure, time '23:59')) at time zone 'UTC') <= b.fin_session union all -- Repas employés : sortie légitime mais qui pèse sur le théorique. select b.id, b.restaurant_id, 'REPAS_EMPLOYE', re.date_heure, round(re.montant::numeric, 2), normaliser_identite(re.vendeur_code), normaliser_identite(re.manager_code), re.beneficiaire, null from v_bornes_session b join repas_employes re on re.restaurant_id = b.restaurant_id and re.caisse = b.caisse and re.date_heure > b.debut_session and re.date_heure <= b.fin_session; grant select on v_bornes_session, v_operations_session to service_role, authenticated, anon; notify pgrst, 'reload schema'; -- --------------------------------------------------------------------- -- CONTRÔLE — ce qui s'est passé pendant les cinq plus gros manquants. -- Une session sans aucune ligne est un résultat, pas un échec. -- --------------------------------------------------------------------- with pires as ( select id, date_fiscale, caisse, badge_code, responsable, ecart_especes from v_bornes_session where restaurant_id = 5 and ecart_especes < -20 order by ecart_especes limit 5) select p.date_fiscale, p.caisse, p.badge_code, p.ecart_especes, o.type, count(o.*) as nb, round(coalesce(sum(o.montant), 0)::numeric, 2) as montant from pires p left join v_operations_session o on o.session_id = p.id group by 1,2,3,4,5 order by p.ecart_especes, o.type;
